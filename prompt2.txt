    /* Audio Synthesis */
    function speakText(text, rate = 0.9) {
      if (!('speechSynthesis' in window) || !text) return;
      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(text);
      utter.lang = 'en-US';
      utter.rate = rate;

      const voices = window.speechSynthesis.getVoices();
      const friendlyVoice = voices.find(v => v.lang.startsWith('en') && (v.name.includes('Samantha') || v.name.includes('Google') || v.name.includes('Natural') || v.name.includes('Victoria')));
      if (friendlyVoice) utter.voice = friendlyVoice;

      window.speechSynthesis.speak(utter);
    }

    /* Pleasant Sound Chimes with Web Audio API */
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playChime(type = 'success') {
      try {
        if (audioCtx.state === 'suspended') audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        const now = audioCtx.currentTime;
        if (type === 'success') {
          osc.frequency.setValueAtTime(523.25, now); // C5
          osc.frequency.exponentialRampToValueAtTime(659.25, now + 0.1); // E5
          osc.frequency.exponentialRampToValueAtTime(783.99, now + 0.2); // G5
          gain.gain.setValueAtTime(0.15, now);
          gain.gain.linearRampToValueAtTime(0, now + 0.35);
          osc.start(now);
          osc.stop(now + 0.35);
        } else if (type === 'pop') {
          osc.frequency.setValueAtTime(400, now);
          osc.frequency.exponentialRampToValueAtTime(800, now + 0.08);
          gain.gain.setValueAtTime(0.1, now);
          gain.gain.linearRampToValueAtTime(0, now + 0.1);
          osc.start(now);
          osc.stop(now + 0.1);
        } else {
          osc.frequency.setValueAtTime(220, now);
          osc.frequency.linearRampToValueAtTime(180, now + 0.15);
          gain.gain.setValueAtTime(0.12, now);
          gain.gain.linearRampToValueAtTime(0, now + 0.2);
          osc.start(now);
          osc.stop(now + 0.2);
        }
      } catch (e) {
        console.warn('Audio Context silent fallback');
      }
    }

    /* Position bubble directly above word with arrow */
    function positionWordBubble(span) {
      const popup = document.getElementById('wordPopup');
      const arrow = document.getElementById('popupArrow');
      const scrollArea = document.getElementById('readerScrollArea');

      const spanRect = span.getBoundingClientRect();
      const scrollRect = scrollArea.getBoundingClientRect();

      popup.classList.remove('hidden');

      const popupWidth = popup.offsetWidth || 265;
      const popupHeight = popup.offsetHeight || 220;

      let left = (spanRect.left - scrollRect.left) + (spanRect.width / 2) - (popupWidth / 2);
      if (left < 10) left = 10;
      if (left + popupWidth > scrollRect.width - 10) left = scrollRect.width - popupWidth - 10;

      const arrowLeft = (spanRect.left - scrollRect.left) + (spanRect.width / 2) - left;
      arrow.style.setProperty('--arrow-left', `${arrowLeft}px`);

      let top = (spanRect.top - scrollRect.top) + scrollArea.scrollTop - popupHeight - 12;
      if (top < scrollArea.scrollTop + 10) {
        top = (spanRect.bottom - scrollRect.top) + scrollArea.scrollTop + 12;
        arrow.style.top = '-9px';
        arrow.style.bottom = 'auto';
        arrow.style.borderTop = 'none';
        arrow.style.borderBottom = '9px solid var(--theme-primary)';
      } else {
        arrow.style.top = 'auto';
        arrow.style.bottom = '-9px';
        arrow.style.borderBottom = 'none';
        arrow.style.borderTop = '9px solid var(--theme-primary)';
      }

      popup.style.left = `${left}px`;
      popup.style.top = `${top}px`;
    }

    async function getWordData(clean, raw) {
      if (CHILD_LEXICON[clean]) {
        return {
          word: clean,
          ipa: CHILD_LEXICON[clean].ipa || `|${clean}|`,
          meaning: CHILD_LEXICON[clean].meaning,
          sentence: CHILD_LEXICON[clean].sentence,
          question: CHILD_LEXICON[clean].question,
          choices: CHILD_LEXICON[clean].choices,
          correct: CHILD_LEXICON[clean].correct
        };
      }

      // Online fallback
      let ipa = `|${clean}|`;
      let meaning = `A wonderful English word used to describe ${clean}.`;
      let sentence = `We can use the word "${clean}" in fun stories.`;
      let question = `Do you know how to use "${clean}"?`;
      let choices = ["Yes, I do!", "Tell me again", "Practice more"];
      let correct = 0;

      try {
        const res = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(clean)}`);
        if (res.ok) {
          const apiData = await res.json();
          if (apiData && apiData[0]) {
            const p = apiData[0].phonetic || (apiData[0].phonetics && apiData[0].phonetics.find(x => x.text)?.text);
            if (p) ipa = `|${p.replace(/\//g, '')}|`;
            const firstM = apiData[0].meanings?.[0];
            if (firstM?.definitions?.[0]?.definition) {
              meaning = firstM.definitions[0].definition;
            }
            if (firstM?.definitions?.[0]?.example) {
              sentence = `"${firstM.definitions[0].example}"`;
            }
          }
        }
      } catch (err) {
        console.warn('Fallback to local definition');
      }

      return { word: clean, ipa, meaning, sentence, question, choices, correct };
    }

          s.classList.add(status === 'learned' ? 'status-learned' : 'status-learning');

  <script>
    /* Synchronized Audiobook Narration */
    function startStoryAudio(startIdx = 0) {
      if (activeWordSpans.length === 0) return;
      appState.isAudioPlaying = true;
      appState.audioWordIdx = startIdx;
      document.getElementById('audioPlayIcon').className = 'fa-solid fa-pause';
      playNextNarratedWord();
    }

    function pauseStoryAudio() {
      appState.isAudioPlaying = false;
      window.speechSynthesis.cancel();
      document.getElementById('audioPlayIcon').className = 'fa-solid fa-play ml-0.5';
      activeWordSpans.forEach(s => s.classList.remove('speaking-now'));
    }

    function playNextNarratedWord() {
      if (!appState.isAudioPlaying || appState.audioWordIdx >= activeWordSpans.length) {
        pauseStoryAudio();
        return;
      }

      activeWordSpans.forEach(s => s.classList.remove('speaking-now'));
      const activeSpan = activeWordSpans[appState.audioWordIdx];
      activeSpan.classList.add('speaking-now');
      activeSpan.scrollIntoView({ behavior: 'smooth', block: 'center' });

      const w = activeSpan.dataset.cleanWord;
      const utter = new SpeechSynthesisUtterance(w);
      utter.lang = 'en-US';
      utter.rate = appState.audioSpeed;

      utter.onend = () => {
        if (!appState.isAudioPlaying) return;
        appState.audioWordIdx++;
        setTimeout(playNextNarratedWord, 110 / appState.audioSpeed);
      };

      utter.onerror = () => {
        if (!appState.isAudioPlaying) return;
        appState.audioWordIdx++;
        setTimeout(playNextNarratedWord, 90);
      };

      window.speechSynthesis.speak(utter);
    }
